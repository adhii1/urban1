\section{Platform Overview}
% ============================================================

TORQQ is a shared mobility platform supporting:
\begin{itemize}
  \item \textbf{Flexi} -- Single-person on-demand rides (book 2 hours ahead, like Ola/Uber)
  \item \textbf{Hybrid} -- 3 days/week shared shuttle with managed bus stops
  \item \textbf{Weekday} -- Mon-Fri daily shared shuttle with managed bus stops
  \item \textbf{Stop-to-Stop} -- All 7 days shared shuttle on fixed routes
\end{itemize}

\subsection{Architecture}
\begin{itemize}
  \item \textbf{Backend}: Node.js + Express + MongoDB + Socket.IO
  \item \textbf{Real-time}: WebSocket namespaces for Customer, Driver, Admin
  \item \textbf{Payment}: Razorpay integration (order creation, verification, refunds)
  \item \textbf{Background Services}: Ride expiry, subscription expiry, scheduled dispatch, trip generation, bundle matching
\end{itemize}

\subsection{Server Configuration}
\begin{lstlisting}
Base URL: http://localhost:4000/api/v1
WebSocket: ws://localhost:4000/sockets/{customer|driver|admin}
Authentication: HTTP-only cookies (JWT access + refresh tokens)
JWT Expiry: 2 hours (access), 7 days (refresh)
\end{lstlisting}

\section{Core Business Flows}
% ============================================================

\subsection{Flexi Ride (Scheduled -- 2 Hours Ahead)}
\begin{enumerate}
  \item Customer sends \texttt{ride:request} with \texttt{scheduledPickupTime} = now + 2 hours
  \item Backend creates RideRequest with status \texttt{SCHEDULED}
  \item \texttt{scheduledDispatchService} checks every minute
  \item 40 minutes before pickup: transitions to \texttt{PENDING}
  \item \texttt{BundleMatchingEngine} finds nearest available driver within 5km
  \item Driver receives \texttt{ride:new-request} with ~40 min to reach pickup
  \item Driver accepts → OTP generated → Customer notified
  \item GPS tracking active throughout
  \item OTP verified at pickup → Ride starts
  \item Driver completes → Final fare calculated
\end{enumerate}

\textbf{Console Log (Verified):}
\begin{lstlisting}

                    Will dispatch ~40min before.
[BUNDLE_DEBUG] New ride request created

  Status: SCHEDULED
  ScheduledPickupTime: 2026-08-10T16:26:46Z

                     closer to your pickup time."
\end{lstlisting}

\subsection{Flexi Ride (Immediate)}
\begin{enumerate}
  \item Customer sends \texttt{ride:request} without scheduledPickupTime
  \item Backend creates RideRequest with status \texttt{PENDING}
  \item \texttt{BundleMatchingEngine} immediately searches for compatible rides
  \item If no bundle forms within 10s, dispatches as single ride
  \item Nearest online+available driver found → offer sent
  \item Driver accepts → full flow (OTP, GPS, completion)
\end{enumerate}

\end{lstlisting}

\subsection{Hybrid/Weekday (Shared Shuttle Bundling)}
\begin{enumerate}
  \item Admin assigns drivers to routes (each route has ordered stops)
  \item Customers subscribe to Hybrid/Weekday plans, pick their route + stops
  \item \texttt{tripGeneratorService} runs daily:
    \begin{itemize}
      \item Finds active subscriptions eligible for today
      \item Groups customers by route
      \item Creates Trip with manifest (all passengers + pickup/drop stops)
      \item Assigns the route's driver
    \end{itemize}
  \item Driver sees scheduled trip with full passenger list
  \item Driver navigates to each stop, verifies OTP per passenger
  \item If customer changes route → auto-moved to new route's trip
\end{enumerate}

\subsection{Customer Route Change}
\begin{lstlisting}
tripGeneratorService.handleRouteChange(customerId, oldRouteId, newRouteId, subscription)
// 1. Removes customer from old route's trip manifest
// 2. Adds to new route's trip (or creates one if none exists)
// 3. New route's assigned driver sees updated manifest
\end{lstlisting}

\section{Payment Integration (Razorpay)}
% ============================================================

\begin{enumerate}
  \item \textbf{Create Order}: Backend calls \texttt{razorpay.orders.create()} with amount in paise
  \item \textbf{Frontend Checkout}: Opens Razorpay popup with order\_id
  \item \textbf{Verify Signature}: Backend verifies HMAC SHA256: \\
    \texttt{hmac(order\_id|payment\_id, secret) === signature}
  \item \textbf{Activate}: On success, subscription status → ACTIVE
  \item \textbf{Refunds}: \texttt{razorpay.payments.refund(payment\_id, amount)}
\end{enumerate}

\textbf{Mock Mode}: When \texttt{RAZORPAY\_KEY\_ID} is not set, all operations return mock responses for testing.

\section{Security \& Infrastructure}
% ============================================================

\begin{itemize}
  \item \textbf{Authentication}: JWT with HTTP-only cookies, auto-refresh
  \item \textbf{Rate Limiting}: Per-IP and per-socket rate limiters
  \item \textbf{CORS}: Whitelisted origins with credentials
  \item \textbf{Helmet}: Full security headers (CSP, HSTS, XSS protection)
  \item \textbf{Input Validation}: Joi schemas on all endpoints
  \item \textbf{Soft Delete}: All models use \texttt{isDeleted} flag
  \item \textbf{Token Blacklist}: Logout invalidates tokens
  \item \textbf{Socket Auth}: JWT verification on WebSocket connection
  \item \textbf{Redis}: Optional adapter for horizontal Socket.IO scaling
\end{itemize}
